# Hugging Face ecosystem.
# transformers >= 5.8 is a hard floor: the `qwen3_5` architecture does not exist
# before it, and an older release fails at load with
#   "checkpoint has model type `qwen3_5` but Transformers does not recognize it".
transformers==5.14.1
accelerate==1.14.0
datasets==5.0.1
huggingface-hub==1.27.0
tokenizers==0.22.2
safetensors==0.8.0

# Quantization. llmcompressor 0.13 is required for the SmoothQuant mapping form
# used here: it resolves mappings through `match_modules_set`, which the exact
# per-layer module names in quantize.py are built for.
llmcompressor==0.13.0
compressed-tensors==0.18.0

# Utilities

pandas==3.0.5

tqdm==4.70.0
